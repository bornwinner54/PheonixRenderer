cmake_minimum_required(VERSION 3.20)

#=========================================================
# Project
#=========================================================

project(
    PhoenixRenderer
    VERSION 0.1.0
    DESCRIPTION "Real-Time Cinematic Rendering Framework"
    LANGUAGES CXX C
)

#=========================================================
# C++ Standard
#=========================================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

#=========================================================
# Build Type
#=========================================================

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug)
endif()

#=========================================================
# Output Directories
#=========================================================

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

#=========================================================
# Project Name
#=========================================================

set(PHOENIX_APP PhoenixRenderer)

#=========================================================
# Build Options
#=========================================================

option(PHOENIX_ENABLE_WARNINGS   "Enable compiler warnings" ON)
option(PHOENIX_ENABLE_IMGUI      "Enable Dear ImGui" ON)
option(PHOENIX_ENABLE_ASSERTS    "Enable assertions" ON)
option(PHOENIX_ENABLE_PROFILING  "Enable profiling" OFF)

#=========================================================
# Source Files
#=========================================================

file(GLOB_RECURSE ENGINE_SOURCES
    CONFIGURE_DEPENDS
    Source/*.cpp
)

file(GLOB_RECURSE ENGINE_HEADERS
    CONFIGURE_DEPENDS
    Source/*.h
)

#=========================================================
# Executable
#=========================================================

add_executable(
    ${PHOENIX_APP}

    ${ENGINE_SOURCES}
    ${ENGINE_HEADERS}
)

#=========================================================
# Include Directories
#=========================================================

set(ENGINE_INCLUDE_DIR
    Source
)

set(THIRD_PARTY_INCLUDE_DIRS

    External/glm

    External/stb

    External/GLEW/include

    External/imgui
)

target_include_directories(

    ${PHOENIX_APP}

    PRIVATE

    ${ENGINE_INCLUDE_DIR}

    ${THIRD_PARTY_INCLUDE_DIRS}

    "C:/glew-2.1.0/include"
    "C:/GLM/g-truc-glm-8d1fd52"
)

#=========================================================
# Compiler Definitions
#=========================================================

target_compile_definitions(

    ${PHOENIX_APP}

    PRIVATE

    WIN32_LEAN_AND_MEAN

    NOMINMAX

    _CRT_SECURE_NO_WARNINGS

    GLEW_STATIC
)

#=========================================================
# Compiler Options
#=========================================================

if(MSVC)

    target_compile_options(

        ${PHOENIX_APP}

        PRIVATE

        /W4          # Warning level

        /permissive- # Standard compliance

        /MP          # Multi-core compilation

        /EHsc
    )

endif()

#=========================================================
# Link Directories
#=========================================================

target_link_directories(

    ${PHOENIX_APP}

    PRIVATE

    External/GLEW/lib
)

link_directories("C:/glew-2.1.0/lib/Release/x64")

#=========================================================
# Libraries
#=========================================================

target_link_libraries(

    ${PHOENIX_APP}

    PRIVATE

    opengl32
    "C:/glew-2.1.0/lib/Release/x64/glew32s.lib"                 # static GLEW (use glew32 if dynamic)
    user32
    gdi32
)

#=========================================================
# Windows Entry Point
#=========================================================

set_target_properties(

    ${PHOENIX_APP}

    PROPERTIES

    WIN32_EXECUTABLE TRUE
)

#=========================================================
# Visual Studio Filters
#=========================================================

source_group(

    TREE

    ${CMAKE_CURRENT_SOURCE_DIR}/Source

    FILES

    ${ENGINE_SOURCES}
    ${ENGINE_HEADERS}
)

#=========================================================
# Copy Assets
#=========================================================

add_custom_command(

    TARGET ${PHOENIX_APP}

    POST_BUILD

    COMMAND ${CMAKE_COMMAND}

    -E copy_directory

    ${CMAKE_SOURCE_DIR}/Assets

    $<TARGET_FILE_DIR:${PHOENIX_APP}>/Assets
)

#=========================================================
# Startup Message
#=========================================================

message(STATUS "")
message(STATUS "======================================")
message(STATUS " Phoenix Renderer Configuration")
message(STATUS "======================================")
message(STATUS "Version        : ${PROJECT_VERSION}")
message(STATUS "Build Type     : ${CMAKE_BUILD_TYPE}")
message(STATUS "C++ Standard   : ${CMAKE_CXX_STANDARD}")
message(STATUS "Compiler       : ${CMAKE_CXX_COMPILER_ID}")
message(STATUS "System         : ${CMAKE_SYSTEM_NAME}")
message(STATUS "======================================")
message(STATUS "")
